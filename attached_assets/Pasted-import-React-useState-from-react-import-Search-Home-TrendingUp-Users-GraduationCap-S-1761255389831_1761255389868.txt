import React, { useState } from 'react';
import { Search, Home, TrendingUp, Users, GraduationCap, Shield, Building, DollarSign, MapPin, AlertCircle, CheckCircle, XCircle, Loader } from 'lucide-react';

const PropertyInvestmentAnalyzer = () => {
  const [address, setAddress] = useState('');
  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState(null);
  const [error, setError] = useState(null);
  const [apiStatus, setApiStatus] = useState({});

  // Parse address to extract ZIP code and state
  const parseAddress = (address) => {
    const zipMatch = address.match(/\b(\d{5})\b/);
    const stateMatch = address.match(/\b([A-Z]{2})\b/i);
    
    return {
      zipCode: zipMatch ? zipMatch[1] : null,
      state: stateMatch ? stateMatch[1].toUpperCase() : null,
      fullAddress: address
    };
  };

  // Get Census data - REAL API
  const getCensusData = async (zipCode) => {
    try {
      const year = 2022;
      const variables = [
        'NAME',
        'B19013_001E', // Median household income
        'B01003_001E', // Total population
        'B23025_003E', // In labor force
        'B23025_005E', // Unemployed
        'B15003_022E', // Bachelor's degree
        'B15003_023E', // Master's degree
        'B15003_024E', // Professional degree
        'B15003_025E', // Doctorate degree
        'B15003_001E', // Total education population
        'B25001_001E'  // Total housing units
      ].join(',');

      const url = `https://api.census.gov/data/${year}/acs/acs5?get=${variables}&for=zip%20code%20tabulation%20area:${zipCode}`;
      
      console.log('Fetching Census data for ZIP:', zipCode);
      
      const response = await fetch(url);
      
      if (!response.ok) {
        throw new Error(`Census API returned ${response.status}`);
      }

      const data = await response.json();
      
      if (!data || data.length < 2) {
        throw new Error('No Census data available for this ZIP code');
      }

      // Parse response - first row is headers, second row is data
      const values = data[1];
      const medianIncome = parseInt(values[1]) || 0;
      const population = parseInt(values[2]) || 0;
      const laborForce = parseInt(values[3]) || 0;
      const unemployed = parseInt(values[4]) || 0;
      const bachelor = parseInt(values[5]) || 0;
      const master = parseInt(values[6]) || 0;
      const professional = parseInt(values[7]) || 0;
      const doctorate = parseInt(values[8]) || 0;
      const eduTotal = parseInt(values[9]) || 1;
      const housingUnits = parseInt(values[10]) || 0;

      const employmentRate = laborForce > 0 ? ((laborForce - unemployed) / laborForce) * 100 : 0;
      const collegeDegreePct = ((bachelor + master + professional + doctorate) / eduTotal) * 100;

      return {
        success: true,
        data: {
          medianIncome,
          population,
          employmentRate: parseFloat(employmentRate.toFixed(1)),
          collegeDegreePct: parseFloat(collegeDegreePct.toFixed(1)),
          housingUnits,
          zipName: values[0]
        }
      };
    } catch (err) {
      console.error('Census API error:', err);
      return {
        success: false,
        error: err.message,
        data: null
      };
    }
  };

  // Get historical growth data (comparing multiple years)
  const getCensusGrowthData = async (zipCode) => {
    try {
      // Fetch 2017 and 2022 data for comparison
      const year1 = 2017;
      const year2 = 2022;
      
      const variables = 'B19013_001E,B01003_001E,B25001_001E'; // Income, Population, Housing
      
      const [response1, response2] = await Promise.all([
        fetch(`https://api.census.gov/data/${year1}/acs/acs5?get=${variables}&for=zip%20code%20tabulation%20area:${zipCode}`),
        fetch(`https://api.census.gov/data/${year2}/acs/acs5?get=${variables}&for=zip%20code%20tabulation%20area:${zipCode}`)
      ]);

      if (!response1.ok || !response2.ok) {
        throw new Error('Could not fetch historical data');
      }

      const data1 = await response1.json();
      const data2 = await response2.json();

      if (!data1 || data1.length < 2 || !data2 || data2.length < 2) {
        throw new Error('Insufficient historical data');
      }

      const income1 = parseInt(data1[1][0]) || 0;
      const income2 = parseInt(data2[1][0]) || 0;
      const pop1 = parseInt(data1[1][1]) || 0;
      const pop2 = parseInt(data2[1][1]) || 0;
      const housing1 = parseInt(data1[1][2]) || 0;
      const housing2 = parseInt(data2[1][2]) || 0;

      const incomeGrowth = income1 > 0 ? ((income2 - income1) / income1) * 100 : 0;
      const popGrowth = pop1 > 0 ? ((pop2 - pop1) / pop1) * 100 : 0;
      const housingGrowth = housing1 > 0 ? ((housing2 - housing1) / housing1) * 100 : 0;

      return {
        success: true,
        data: {
          populationGrowth5yr: parseFloat(popGrowth.toFixed(1)),
          incomeGrowth5yr: parseFloat(incomeGrowth.toFixed(1)),
          housingGrowth5yr: parseFloat(housingGrowth.toFixed(1))
        }
      };
    } catch (err) {
      console.error('Census growth data error:', err);
      return {
        success: false,
        error: err.message,
        data: {
          populationGrowth5yr: 5.0,
          incomeGrowth5yr: 8.0,
          housingGrowth5yr: 4.0
        }
      };
    }
  };

  // Calculate walkability from OpenStreetMap - REAL API
  const calculateWalkability = async (address) => {
    try {
      // Step 1: Geocode the address using Nominatim (OSM)
      const nominatimUrl = `https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(address)}&limit=1`;
      
      console.log('Geocoding address with Nominatim...');
      
      const geoResponse = await fetch(nominatimUrl, {
        headers: {
          'User-Agent': 'PropertyInvestmentAnalyzer/1.0 (Educational Tool)'
        }
      });

      if (!geoResponse.ok) {
        throw new Error('Geocoding failed');
      }

      const geoData = await geoResponse.json();
      
      if (!geoData || geoData.length === 0) {
        throw new Error('Address not found');
      }

      const lat = parseFloat(geoData[0].lat);
      const lon = parseFloat(geoData[0].lon);

      console.log('Address geocoded to:', lat, lon);

      // Step 2: Query Overpass API for nearby amenities within 1km
      const overpassUrl = 'https://overpass-api.de/api/interpreter';
      
      // Count different types of amenities
      const query = `
        [out:json][timeout:25];
        (
          node["shop"="supermarket"](around:1000,${lat},${lon});
          node["shop"="convenience"](around:1000,${lat},${lon});
          node["amenity"="restaurant"](around:1000,${lat},${lon});
          node["amenity"="cafe"](around:1000,${lat},${lon});
          node["amenity"="school"](around:1000,${lat},${lon});
          node["amenity"="library"](around:1000,${lat},${lon});
          node["leisure"="park"](around:1000,${lat},${lon});
          node["amenity"="bank"](around:1000,${lat},${lon});
          node["amenity"="pharmacy"](around:1000,${lat},${lon});
          node["amenity"="hospital"](around:1000,${lat},${lon});
          node["amenity"="clinic"](around:1000,${lat},${lon});
          node["public_transport"="station"](around:1000,${lat},${lon});
          node["public_transport"="stop_position"](around:1000,${lat},${lon});
          node["highway"="bus_stop"](around:1000,${lat},${lon});
        );
        out count;
      `;

      console.log('Querying Overpass API for nearby amenities...');

      const poiResponse = await fetch(overpassUrl, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/x-www-form-urlencoded'
        },
        body: `data=${encodeURIComponent(query)}`
      });

      if (!poiResponse.ok) {
        throw new Error('Amenity query failed');
      }

      const poiData = await poiResponse.json();
      const amenityCount = poiData.elements ? poiData.elements.length : 0;

      console.log('Found amenities:', amenityCount);

      // Calculate walk score (0-100) based on amenity density
      // This is a simplified version of Walk Score's algorithm
      // More amenities = higher score, capped at 100
      let walkScore = Math.min(100, Math.round(amenityCount * 1.5));
      
      // Adjust score based on density tiers
      if (amenityCount < 5) walkScore = Math.min(walkScore, 30);
      else if (amenityCount < 15) walkScore = Math.min(walkScore, 60);
      else if (amenityCount < 30) walkScore = Math.min(walkScore, 80);
      
      return {
        success: true,
        data: {
          walkScore,
          transitScore: Math.round(walkScore * 0.85), // Approximation
          bikeScore: Math.round(walkScore * 0.90), // Approximation
          amenityCount,
          lat,
          lon
        }
      };
    } catch (err) {
      console.error('Walkability calculation error:', err);
      return {
        success: false,
        error: err.message,
        data: {
          walkScore: 0,
          transitScore: 0,
          bikeScore: 0,
          amenityCount: 0,
          lat: null,
          lon: null
        }
      };
    }
  };

  // Main analysis function
  const analyzeProperty = async (inputAddress) => {
    setLoading(true);
    setError(null);
    setApiStatus({
      census: null,
      growth: null,
      walkability: null
    });

    try {
      const parsed = parseAddress(inputAddress);
      
      if (!parsed.zipCode) {
        throw new Error('Please include a 5-digit ZIP code in the address');
      }

      console.log('Starting analysis for:', inputAddress);

      // Call real APIs in parallel
      setApiStatus({ census: 'loading', growth: 'loading', walkability: 'loading' });
      
      const [censusResult, growthResult, walkResult] = await Promise.all([
        getCensusData(parsed.zipCode),
        getCensusGrowthData(parsed.zipCode),
        calculateWalkability(inputAddress)
      ]);

      // Update API status
      setApiStatus({
        census: censusResult.success ? 'success' : 'error',
        growth: growthResult.success ? 'success' : 'error',
        walkability: walkResult.success ? 'success' : 'error'
      });

      // Check if we have enough data to proceed
      if (!censusResult.data) {
        throw new Error('Unable to retrieve Census data for this ZIP code');
      }

      // Simulated property and school data (requires paid APIs or scraping)
      const propertyData = {
        price: 425000,
        bedrooms: 3,
        bathrooms: 2,
        sqft: 1850,
        yearBuilt: 2008,
        pricePerSqft: 230,
        estimatedRent: 2400
      };

      const schoolData = {
        elementary: 8,
        middle: 7,
        high: 9,
        average: 8.0
      };

      const marketData = {
        appreciationRate3yr: 18.5,
        daysOnMarket: 28,
        priceToRentRatio: 14.7,
        rentalYield: 6.8,
        vacancyRate: 4.2
      };

      // For demo crime data (FBI API has complex authentication)
      const crimeData = {
        crimeScore: 7.2,
        trend: 'improving',
        violentCrimeRate: 3.2,
        propertyCrimeRate: 18.5
      };

      // Combine all metrics
      const metrics = {
        property: propertyData,
        location: walkResult.data,
        schools: schoolData,
        crime: crimeData,
        demographics: {
          ...censusResult.data,
          ...growthResult.data
        },
        market: marketData
      };

      // Calculate scores
      const scores = {
        marketStrength: calculateMarketScore(metrics.market),
        demographicTrends: calculateDemographicScore(metrics.demographics),
        schoolQuality: (metrics.schools.average / 10) * 100,
        safetyScore: metrics.crime.crimeScore * 10,
        walkability: (metrics.location.walkScore / 100) * 100,
        rentalPotential: calculateRentalScore(metrics.market)
      };

      const weights = {
        marketStrength: 0.25,
        demographicTrends: 0.20,
        schoolQuality: 0.20,
        safetyScore: 0.15,
        walkability: 0.10,
        rentalPotential: 0.10
      };

      const finalScore = Object.keys(scores).reduce((total, key) => {
        return total + (scores[key] * weights[key]);
      }, 0);

      setResult({
        address: inputAddress,
        zipCode: parsed.zipCode,
        state: parsed.state,
        finalScore: (finalScore / 10).toFixed(1),
        metrics,
        scores,
        weights,
        reasoning: generateReasoning(metrics, scores, finalScore / 10),
        dataSource: {
          census: censusResult.success ? 'Live Census API ✓' : 'Census API Error',
          growth: growthResult.success ? 'Live Census API ✓' : 'Demo Growth Data',
          walkability: walkResult.success ? 'Live OpenStreetMap ✓' : 'OSM Error',
          crime: 'Demo Data (FBI API requires auth)',
          property: 'Demo Data (requires paid API)',
          schools: 'Demo Data (requires paid API)'
        }
      });
      
    } catch (err) {
      setError(err.message);
      console.error('Analysis error:', err);
    } finally {
      setLoading(false);
    }
  };

  const calculateMarketScore = (market) => {
    let score = 0;
    score += market.appreciationRate3yr > 15 ? 30 : market.appreciationRate3yr > 10 ? 20 : 10;
    score += market.daysOnMarket < 30 ? 30 : market.daysOnMarket < 60 ? 20 : 10;
    score += market.priceToRentRatio < 15 ? 30 : market.priceToRentRatio < 20 ? 20 : 10;
    score += market.vacancyRate < 5 ? 10 : 5;
    return score;
  };

  const calculateDemographicScore = (demo) => {
    let score = 0;
    score += demo.populationGrowth5yr > 7 ? 25 : demo.populationGrowth5yr > 4 ? 18 : demo.populationGrowth5yr > 0 ? 10 : 5;
    score += demo.incomeGrowth5yr > 10 ? 25 : demo.incomeGrowth5yr > 5 ? 18 : demo.incomeGrowth5yr > 0 ? 10 : 5;
    score += demo.employmentRate > 95 ? 20 : demo.employmentRate > 90 ? 15 : demo.employmentRate > 85 ? 10 : 5;
    score += demo.collegeDegreePct > 40 ? 15 : demo.collegeDegreePct > 30 ? 10 : demo.collegeDegreePct > 20 ? 5 : 2;
    score += demo.housingGrowth5yr > 5 ? 15 : demo.housingGrowth5yr > 3 ? 10 : demo.housingGrowth5yr > 0 ? 5 : 2;
    return score;
  };

  const calculateRentalScore = (market) => {
    let score = 0;
    score += market.rentalYield > 6 ? 50 : market.rentalYield > 4 ? 35 : 20;
    score += market.vacancyRate < 5 ? 30 : market.vacancyRate < 7 ? 20 : 10;
    score += market.priceToRentRatio < 15 ? 20 : 10;
    return score;
  };

  const generateReasoning = (metrics, scores, finalScore) => {
    const reasons = [];
    
    if (scores.marketStrength >= 70) {
      reasons.push(`Strong market fundamentals with ${metrics.market.appreciationRate3yr}% appreciation over 3 years and ${metrics.market.daysOnMarket} days on market.`);
    }
    
    if (scores.demographicTrends >= 70) {
      reasons.push(`Strong demographic trends: ${metrics.demographics.populationGrowth5yr}% population growth and ${metrics.demographics.incomeGrowth5yr}% income growth indicate rising demand.`);
    } else if (metrics.demographics.populationGrowth5yr < 0) {
      reasons.push(`Population declining by ${Math.abs(metrics.demographics.populationGrowth5yr)}% - consider long-term demand carefully.`);
    }
    
    if (scores.schoolQuality >= 75) {
      reasons.push(`Top-rated schools (avg ${metrics.schools.average}/10) make this area attractive to families.`);
    }
    
    if (scores.safetyScore >= 70) {
      reasons.push(`Safety score of ${metrics.crime.crimeScore}/10 with ${metrics.crime.trend} trend.`);
    }
    
    if (metrics.location.walkScore >= 70) {
      reasons.push(`High walkability score of ${metrics.location.walkScore}/100 with ${metrics.location.amenityCount} nearby amenities enhances lifestyle appeal.`);
    } else if (metrics.location.walkScore < 50) {
      reasons.push(`Lower walkability (${metrics.location.walkScore}/100) - car-dependent area may limit appeal to some demographics.`);
    }
    
    if (metrics.market.rentalYield >= 6) {
      reasons.push(`Strong rental yield of ${metrics.market.rentalYield}% with estimated monthly rent of $${metrics.property.estimatedRent.toLocaleString()}.`);
    }
    
    if (metrics.demographics.employmentRate > 95) {
      reasons.push(`High employment rate of ${metrics.demographics.employmentRate}% indicates economic stability.`);
    }

    if (finalScore < 6) {
      reasons.push(`Consider market conditions and growth trends carefully before investing.`);
    }

    return reasons;
  };

  const getScoreColor = (score) => {
    if (score >= 8) return 'text-green-600';
    if (score >= 6.5) return 'text-blue-600';
    if (score >= 5) return 'text-yellow-600';
    return 'text-red-600';
  };

  const getScoreBgColor = (score) => {
    if (score >= 8) return 'bg-green-100';
    if (score >= 6.5) return 'bg-blue-100';
    if (score >= 5) return 'bg-yellow-100';
    return 'bg-red-100';
  };

  return (
    <div className="min-h-screen bg-gradient-to-br from-blue-50 to-indigo-100 p-6">
      <div className="max-w-6xl mx-auto">
        <div className="bg-white rounded-2xl shadow-xl p-8 mb-6">
          <div className="flex items-center gap-3 mb-6">
            <Home className="w-8 h-8 text-indigo-600" />
            <h1 className="text-3xl font-bold text-gray-800">Property Investment Analyzer</h1>
          </div>
          
          <div className="bg-green-50 border-l-4 border-green-500 p-4 mb-4">
            <div className="flex gap-2">
              <CheckCircle className="w-5 h-5 text-green-600 flex-shrink-0 mt-0.5" />
              <div className="text-sm text-green-900">
                <p className="font-semibold mb-1">Live APIs Connected!</p>
                <p>Using US Census Bureau API and OpenStreetMap/Overpass API for real-time data</p>
              </div>
            </div>
          </div>

          <div className="bg-blue-50 border-l-4 border-blue-500 p-4 mb-6">
            <div className="flex gap-2">
              <AlertCircle className="w-5 h-5 text-blue-600 flex-shrink-0 mt-0.5" />
              <div className="text-sm text-blue-900">
                <p className="font-semibold mb-1">Fair Housing Compliant</p>
                <p>Uses only economic indicators, market trends, and public safety data</p>
              </div>
            </div>
          </div>

          <div className="flex gap-3">
            <input
              type="text"
              placeholder="Enter full address with ZIP (e.g., 123 Main St, Austin, TX 78701)"
              value={address}
              onChange={(e) => setAddress(e.target.value)}
              onKeyPress={(e) => e.key === 'Enter' && address && !loading && analyzeProperty(address)}
              className="flex-1 px-4 py-3 border-2 border-gray-300 rounded-lg focus:border-indigo-500 focus:outline-none"
            />
            <button
              onClick={() => analyzeProperty(address)}
              disabled={!address || loading}
              className="px-6 py-3 bg-indigo-600 text-white rounded-lg font-semibold hover:bg-indigo-700 disabled:bg-gray-400 disabled:cursor-not-allowed flex items-center gap-2"
            >
              <Search className="w-5 h-5" />
              {loading ? 'Analyzing...' : 'Analyze'}
            </button>
          </div>

          {error && (
            <div className="mt-4 bg-red-50 border-l-4 border-red-500 p-4">
              <div className="flex gap-2">
                <XCircle className="w-5 h-5 text-red-600 flex-shrink-0 mt-0.5" />
                <div className="text-sm text-red-900">
                  <p className="font-semibold">Error</p>
                  <p>{error}</p>
                </div>
              </div>
            </div>
          )}
        </div>

        {loading && (
          <div className="bg-white rounded-2xl shadow-xl p-12 text-center">
            <div className="animate-spin rounded-full h-16 w-16 border-b-4 border-indigo-600 mx-auto mb-4"></div>
            <p className="text-gray-600 mb-4">Fetching live data from Census Bureau and OpenStreetMap...</p>
            <div className="mt-4 space-y-2 text-sm text-gray-500">
              <div className="flex items-center justify-center gap-2">
                {apiStatus.census === 'loading' && <Loader className="w-4 h-4 animate-spin text-blue-600" />}
                {apiStatus.census === 'success' && <CheckCircle className="w-4 h-4 text-green-600" />}
                {apiStatus.census === 'error' && <XCircle className="w-4 h-4 text-red-600" />}
                <span>Census Demographics</span>
              </div>
              <div className="flex items-center justify-center gap-2">
                {apiStatus.growth === 'loading' && <Loader className="w-4 h-4 animate-spin text-blue-600" />}
                {apiStatus.growth === 'success' && <CheckCircle className="w-4 h-4 text-green-600" />}
                {apiStatus.growth === 'error' && <XCircle className="w-4 h-4 text-red-600" />}
                <span>Historical Growth Data</span>
              </div>
              <div className="flex items-center justify-center gap-2">
                {apiStatus.walkability === 'loading' && <Loader className="w-4 h-4 animate-spin text-blue-600" />}
                {apiStatus.walkability === 'success' && <CheckCircle className="w-4 h-4 text-green-600" />}
                {apiStatus.walkability === 'error' && <XCircle className="w-4 h-4 text-red-600" />}
                <span>Walkability Analysis</span>
              </div>
            </div>
          </div>
        )}

        {result && !loading && (
          <div className="space-y-6">
            <div className="bg-white rounded-xl shadow-lg p-4">
              <h3 className="font-semibold mb-2 text-sm text-gray-700">Data Sources:</h3>
              <div className="grid grid-cols-2 gap-2 text-xs">
                <div className={`px-3 py-1 rounded ${result.dataSource.census.includes('✓') ? 'bg-green-100 text-green-800' : 'bg-gray-100 text-gray-800'}`}>
                  Demographics: {result.dataSource.census}
                </div>
                <div className={`px-3 py-1 rounded ${result.dataSource.growth.includes('✓') ? 'bg-green-100 text-green-800' : 'bg-gray-100 text-gray-800'}`}>
                  Growth: {result.dataSource.growth}
                </div>
                <div className={`px-3 py-1 rounded ${result.dataSource.walkability.includes('✓') ? 'bg-green-100 text-green-800' : 'bg-gray-100 text-gray-800'}`}>
                  Walkability: {result.dataSource.walkability}
                </div>
                <div className="bg-gray-100 text-gray-800 px-3 py-1 rounded">
                  Crime: {result.dataSource.crime}
                </div>
                <div className="bg-gray-100 text-gray-800 px-3 py-1 rounded">
                  Property: {result.dataSource.property}
                </div>
                <div className="bg-gray-100 text-gray-800 px-3 py-1 rounded">
                  Schools: {result.dataSource.schools}
                </div>
              </div>
            </div>

            <div className="bg-white rounded-2xl shadow-xl p-8">
              <div className="flex items-center justify-between mb-6">
                <div>
                  <h2 className="text-2xl font-bold text-gray-800 mb-2">Investment Score</h2>
                  <p className="text-gray-600 flex items-center gap-2">
                    <MapPin className="w-4 h-4" />
                    {result.address}
                  </p>
                  <p className="text-sm text-gray-500">ZIP: {result.zipCode} {result.state && `| State: ${result.state}`}</p>
                  {result.metrics.demographics.zipName && (
                    <p className="text-xs text-gray-400 mt-1">{result.metrics.demographics.zipName}</p>
                  )}
                </div>
                <div className={`text-6xl font-bold ${getScoreColor(result.finalScore)}`}>
                  {result.finalScore}/10
                </div>
              </div>

              <div className="grid grid-cols-2 md:grid-cols-3 gap-4 mb-6">
                <div className={`${getScoreBgColor(result.scores.marketStrength / 10)} rounded-lg p-4`}>
                  <div className="flex items-center gap-2 mb-2">
                    <TrendingUp className="w-5 h-5 text-gray-700" />
                    <span className="font-semibold text-gray-800">Market</span>
                  </div>
                  <div className="text-2xl font-bold text-gray-800">{(result.scores.marketStrength / 10).toFixed(1)}</div>
                </div>

                <div className={`${getScoreBgColor(result.scores.demographicTrends / 10)} rounded-lg p-4`}>
                  <div className="flex items-center gap-2 mb-2">
                    <Users className="w-5 h-5 text-gray-700" />
                    <span className="font-semibold text-gray-800">Demographics</span>
                  </div>
                  <div className="text-2xl font-bold text-gray-800">{(result.scores.demographicTrends / 10).toFixed(1)}</div>
                </div>

                <div className={`${getScoreBgColor(result.scores.schoolQuality / 10)} rounded-lg p-4`}>
                  <div className="flex items-center gap-2 mb-2">
                    <GraduationCap className="w-5 h-5 text-gray-700" />
                    <span className="font-semibold text-gray-800">Schools</span>
                  </div>
